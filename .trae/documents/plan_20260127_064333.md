#### 2. 美化 Observation 渲染 - 新增辅助方法

```python
def _format_action_result(self, result: Dict[str, Any]) -> tuple[str, str, str]:
    """提取可读内容，返回 (display_text, border_style, title)"""
    if not isinstance(result, dict):
        return str(result), "dim", "📥 Observation"
    
    stdout = result.get('output', {}).get('stdout') or result.get('stdout') or ""
    stderr = result.get('output', {}).get('stderr') or result.get('stderr') or ""
    msg = result.get('message') or ""
    status = result.get('status', 'unknown')
    
    display_parts = []
    if stdout:
        display_parts.append(f"{stdout}")
    if stderr:
        display_parts.append(f"[bold red]{stderr}[/bold red]")
    if msg and not stdout:
        display_parts.append(f"[cyan]{msg}[/cyan]")
    
    if not display_parts:
        display_parts.append(str(result))
    
    display_text = "\n".join(display_parts)
    is_error = status == "error" or "error" in str(stderr).lower()
    border_style = "bold red" if is_error else "blue"
    title_suffix = " [ERROR]" if is_error else ""
    
    return display_text.strip(), border_style, f"📥 Observation{title_suffix}"

def _should_highlight_shell(self, action_name: str, content: str) -> bool:
    """判断是否需要对输出进行 shell 语法高亮"""
    shell_actions = {"execute_shell", "run_in_conda_env", "run_python_script", "run_shell"}
    is_shell = action_name in shell_actions or "shell" in action_name.lower()
    has_newline = "\n" in content
    return is_shell and has_newline
```
